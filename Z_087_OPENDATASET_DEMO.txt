*&---------------------------------------------------------------------*
*& Report Z_087_OPENDATASET_DEMO
*&---------------------------------------------------------------------*
*& INPUTテキストファイルを読み込み、同じ内容をOUTPUTへ書き込む
*& 学習用デモプログラム
*&---------------------------------------------------------------------*
REPORT z_087_opendataset_demo.

*---------------------------------------------------------------------*
* 型・データ定義
*---------------------------------------------------------------------*
TYPES: ty_t_line TYPE STANDARD TABLE OF string WITH EMPTY KEY.

DATA: gt_line          TYPE ty_t_line,
      gv_read_count    TYPE i,
      gv_write_count   TYPE i,
      gv_process_type  TYPE string,
      gv_error_message TYPE string.

*---------------------------------------------------------------------*
* Selection Screen
*---------------------------------------------------------------------*
" ファイル入出力設定
SELECTION-SCREEN BEGIN OF BLOCK b01 WITH FRAME.
SELECTION-SCREEN BEGIN OF LINE.
PARAMETERS p_local RADIOBUTTON GROUP r01 DEFAULT 'X' USER-COMMAND md.
SELECTION-SCREEN COMMENT 5(10) text-001 FOR FIELD p_local.
PARAMETERS p_server RADIOBUTTON GROUP r01.
SELECTION-SCREEN COMMENT 30(10) text-002 FOR FIELD p_server.
SELECTION-SCREEN END OF LINE.
SELECTION-SCREEN SKIP.
SELECTION-SCREEN BEGIN OF LINE.
SELECTION-SCREEN COMMENT 1(20) text-003 FOR FIELD p_infile.
PARAMETERS p_infile TYPE rlgrap-filename LOWER CASE OBLIGATORY.
SELECTION-SCREEN END OF LINE.
SELECTION-SCREEN BEGIN OF LINE.
SELECTION-SCREEN COMMENT 1(20) text-004 FOR FIELD p_outfil.
PARAMETERS p_outfil TYPE rlgrap-filename LOWER CASE OBLIGATORY.
SELECTION-SCREEN END OF LINE.
SELECTION-SCREEN END OF BLOCK b01.

AT SELECTION-SCREEN ON VALUE-REQUEST FOR p_infile.
  " ローカル処理時のみ、SAP GUIのファイル選択ダイアログを表示する
  IF p_local = abap_true.
  PERFORM select_local_file CHANGING p_infile.
  ENDIF.

START-OF-SELECTION.
  CLEAR: gt_line, gv_read_count, gv_write_count, gv_error_message.

  IF p_local = abap_true.
    gv_process_type = 'LOCAL'.
    PERFORM process_local.
  ELSEIF p_server = abap_true.
    gv_process_type = 'SERVER'.
    PERFORM process_server.
  ENDIF.

  IF gv_error_message IS INITIAL.
    PERFORM display_result.
  ELSE.
    " エラー時は結果表示ではなく、具体的なメッセージを表示する
    MESSAGE gv_error_message TYPE 'S' DISPLAY LIKE 'E'.
  ENDIF.

*---------------------------------------------------------------------*
* ローカルファイル処理
*---------------------------------------------------------------------*
FORM process_local.
  DATA lv_exists TYPE abap_bool.

  " INPUTファイルの存在を確認する。存在しない場合はOUTPUTを作成しない
  PERFORM check_local_file USING p_infile CHANGING lv_exists.
  IF lv_exists = abap_false.
    IF gv_error_message IS INITIAL.
      gv_error_message = |INPUTファイルが存在しません。: { p_infile }|.
    ENDIF.
    RETURN.
  ENDIF.

  " ローカルファイルを読み込み、内部テーブルに格納する
  PERFORM read_local_file.
  IF gv_error_message IS NOT INITIAL.
    RETURN.
  ENDIF.

  " 内部テーブルの内容をローカルOUTPUTファイルへ書き込む
  PERFORM write_local_file.
ENDFORM.

*---------------------------------------------------------------------*
* ローカルINPUTファイルの存在確認
*---------------------------------------------------------------------*
FORM check_local_file USING    iv_filename TYPE rlgrap-filename
                      CHANGING cv_exists   TYPE abap_bool.
  DATA lv_filename TYPE string.

  CLEAR cv_exists.
  lv_filename = iv_filename.

  CALL METHOD cl_gui_frontend_services=>file_exist
    EXPORTING
      file                 = lv_filename
    RECEIVING
      result               = cv_exists
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      wrong_parameter      = 3
      not_supported_by_gui = 4
      OTHERS               = 5.
  IF sy-subrc <> 0.
    gv_error_message = |ローカルINPUTファイルの存在確認に失敗しました。(RC={ sy-subrc })|.
  ENDIF.
ENDFORM.

*---------------------------------------------------------------------*
* ローカルファイル読み込み（SAP GUIフロントエンド機能を使用）
*---------------------------------------------------------------------*
FORM read_local_file.
  DATA lv_filename TYPE string.

  lv_filename = p_infile.

  CALL METHOD cl_gui_frontend_services=>gui_upload
    EXPORTING
      filename                = lv_filename
      filetype                = 'ASC'
    CHANGING
      data_tab                = gt_line
    EXCEPTIONS
      file_open_error         = 1
      file_read_error         = 2
      no_batch                = 3
      gui_refuse_filetransfer = 4
      invalid_type            = 5
      no_authority            = 6
      unknown_error           = 7
      bad_data_format         = 8
      header_not_allowed      = 9
      separator_not_allowed   = 10
      header_too_long         = 11
      unknown_dp_error        = 12
      access_denied           = 13
      dp_out_of_memory        = 14
      disk_full               = 15
      dp_timeout              = 16
      not_supported_by_gui    = 17
      error_no_gui            = 18
      OTHERS                  = 19.
  IF sy-subrc <> 0.
    gv_error_message = |ローカルINPUTファイルを読み込めません。(RC={ sy-subrc })|.
    RETURN.
  ENDIF.

  gv_read_count = lines( gt_line ).
ENDFORM.

*---------------------------------------------------------------------*
* ローカルOUTPUTファイル書き込み（既存ファイルは上書き）
*---------------------------------------------------------------------*
FORM write_local_file.
  DATA lv_filename TYPE string.

  lv_filename = p_outfil.

  CALL METHOD cl_gui_frontend_services=>gui_download
    EXPORTING
      filename                = lv_filename
      filetype                = 'ASC'
    CHANGING
      data_tab                = gt_line
    EXCEPTIONS
      file_write_error        = 1
      no_batch                = 2
      gui_refuse_filetransfer = 3
      invalid_type            = 4
      no_authority            = 5
      unknown_error           = 6
      header_not_allowed      = 7
      separator_not_allowed   = 8
      filesize_not_allowed    = 9
      header_too_long         = 10
      dp_error_create         = 11
      dp_error_send           = 12
      dp_error_write          = 13
      unknown_dp_error        = 14
      access_denied           = 15
      dp_out_of_memory        = 16
      disk_full               = 17
      dp_timeout              = 18
      file_not_found          = 19
      dataprovider_exception  = 20
      control_flush_error     = 21
      not_supported_by_gui    = 22
      error_no_gui            = 23
      OTHERS                  = 24.
  IF sy-subrc <> 0.
    gv_error_message = |ローカルOUTPUTファイルを作成または書き込めません。(RC={ sy-subrc })|.
    RETURN.
  ENDIF.

  gv_write_count = lines( gt_line ).
ENDFORM.

*---------------------------------------------------------------------*
* アプリケーションサーバーファイル処理
*---------------------------------------------------------------------*
FORM process_server.
  DATA lv_exists TYPE abap_bool.

  " OPEN DATASETでINPUTファイルの存在およびアクセス可否を確認する
  PERFORM check_server_file USING p_infile CHANGING lv_exists.
  IF lv_exists = abap_false.
    IF gv_error_message IS INITIAL.
      gv_error_message = |INPUTファイルが存在しません。: { p_infile }|.
    ENDIF.
    RETURN.
  ENDIF.

  " アプリケーションサーバー上のINPUTファイルを1行ずつ読み込む
  PERFORM read_server_file.
  IF gv_error_message IS NOT INITIAL.
    RETURN.
  ENDIF.

  " アプリケーションサーバー上のOUTPUTファイルへ1行ずつ書き込む
  PERFORM write_server_file.
ENDFORM.

*---------------------------------------------------------------------*
* サーバーINPUTファイルの存在・OPEN可否確認
*---------------------------------------------------------------------*
FORM check_server_file USING    iv_filename TYPE rlgrap-filename
                       CHANGING cv_exists   TYPE abap_bool.
  CLEAR cv_exists.

  OPEN DATASET iv_filename FOR INPUT IN TEXT MODE ENCODING DEFAULT.
  IF sy-subrc <> 0.
    gv_error_message = |サーバーINPUTファイルが存在しない、またはOPENできません。(RC={ sy-subrc })|.
    RETURN.
  ENDIF.

  cv_exists = abap_true.
  CLOSE DATASET iv_filename.
  IF sy-subrc <> 0.
    gv_error_message = |サーバーINPUTファイルをCLOSEできません。(RC={ sy-subrc })|.
    CLEAR cv_exists.
  ENDIF.
ENDFORM.

*---------------------------------------------------------------------*
* アプリケーションサーバーファイル読み込み
*---------------------------------------------------------------------*
FORM read_server_file.
  DATA lv_line TYPE string.

  OPEN DATASET p_infile FOR INPUT IN TEXT MODE ENCODING DEFAULT.
  IF sy-subrc <> 0.
    gv_error_message = |サーバーINPUTファイルをOPENできません。(RC={ sy-subrc })|.
    RETURN.
  ENDIF.

  DO.
    CLEAR lv_line.
    READ DATASET p_infile INTO lv_line.
    IF sy-subrc = 0.
      APPEND lv_line TO gt_line.
      gv_read_count = gv_read_count + 1.
    ELSEIF sy-subrc = 4.
      " ファイル終端
      EXIT.
    ELSE.
      gv_error_message = |サーバーINPUTファイルの読み込みに失敗しました。(RC={ sy-subrc })|.
      EXIT.
    ENDIF.
  ENDDO.

  CLOSE DATASET p_infile.
  IF sy-subrc <> 0 AND gv_error_message IS INITIAL.
    gv_error_message = |サーバーINPUTファイルをCLOSEできません。(RC={ sy-subrc })|.
  ENDIF.
ENDFORM.

*---------------------------------------------------------------------*
* アプリケーションサーバーOUTPUTファイル書き込み
*---------------------------------------------------------------------*
FORM write_server_file.
  DATA lv_line TYPE string.

  " FOR OUTPUTはファイルがなければ新規作成し、あれば既存内容を上書きする
  OPEN DATASET p_outfil FOR OUTPUT IN TEXT MODE ENCODING DEFAULT.
  IF sy-subrc <> 0.
    gv_error_message = |サーバーOUTPUTファイルを作成またはOPENできません。(RC={ sy-subrc })|.
    RETURN.
  ENDIF.

  LOOP AT gt_line INTO lv_line.
    TRANSFER lv_line TO p_outfil.
    IF sy-subrc <> 0.
      gv_error_message = |サーバーOUTPUTファイルへの書き込みに失敗しました。(RC={ sy-subrc })|.
      EXIT.
    ENDIF.
    gv_write_count = gv_write_count + 1.
  ENDLOOP.

  CLOSE DATASET p_outfil.
  IF sy-subrc <> 0 AND gv_error_message IS INITIAL.
    gv_error_message = |サーバーOUTPUTファイルをCLOSEできません。(RC={ sy-subrc })|.
  ENDIF.
ENDFORM.

*---------------------------------------------------------------------*
* 正常終了時の処理結果表示
*---------------------------------------------------------------------*
FORM display_result.
  WRITE: / '処理方式       ：', gv_process_type,
         / 'INPUTファイル  ：', p_infile,
         / 'OUTPUTファイル ：', p_outfil,
         / '読み込み行数   ：', gv_read_count,
         / '書き込み行数   ：', gv_write_count,
         / '処理結果       ：正常終了'.
ENDFORM.

*---------------------------------------------------------------------*
* ローカルファイル選択ダイアログ
*---------------------------------------------------------------------*
FORM select_local_file CHANGING cv_filename TYPE rlgrap-filename.
  DATA: lt_file_table TYPE filetable,
        ls_file       TYPE LINE OF filetable,
        lv_rc         TYPE i,
        lv_action     TYPE i,
        lv_title      TYPE string,
        lv_extension  TYPE string,
        lv_filter     TYPE string.

  lv_title     = 'ファイルを選択してください'.
  lv_extension = 'txt'.
  lv_filter    = 'Text files (*.txt)|*.txt|All files (*.*)|*.*|'.

  CALL METHOD cl_gui_frontend_services=>file_open_dialog
    EXPORTING
      window_title      = lv_title
      default_extension = lv_extension
      file_filter       = lv_filter
    CHANGING
      file_table        = lt_file_table
      rc                = lv_rc
      user_action       = lv_action
    EXCEPTIONS
      file_open_dialog_failed = 1
      cntl_error              = 2
      error_no_gui            = 3
      not_supported_by_gui    = 4
      OTHERS                  = 5.
  IF sy-subrc = 0 AND lv_action = cl_gui_frontend_services=>action_ok AND lv_rc > 0.
    READ TABLE lt_file_table INDEX 1 INTO ls_file.
    IF sy-subrc = 0.
      cv_filename = ls_file-filename.
    ENDIF.
  ENDIF.
ENDFORM.
