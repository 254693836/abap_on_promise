REPORT Z_087_PROGRAM_ANALYSIS.
"----------------------------------------------------------------------
" 指定したZ/Yレポートの処理単位と呼出し関係を表示する解析ツール
" 対象プログラムのソースは参照するだけで、更新は一切行わない。
"----------------------------------------------------------------------

TABLES TRDIR.

PARAMETERS P_PROG TYPE PROGNAME OBLIGATORY.

TYPES: BEGIN OF TY_SOURCE,
         PROGRAM TYPE PROGNAME,
         LINE    TYPE STRING,
       END OF TY_SOURCE,
       BEGIN OF TY_NODE,
         INDEX       TYPE I,
         KIND        TYPE CHAR12,
         NAME        TYPE STRING,
         DESCRIPTION TYPE STRING,
         PROGRAM     TYPE PROGNAME,
         FIRST_LINE  TYPE I,
         LAST_LINE   TYPE I,
       END OF TY_NODE,
       BEGIN OF TY_CALL,
         CALLER_INDEX TYPE I,
         CALL_KIND    TYPE CHAR12,
         TARGET_NAME  TYPE STRING,
         TARGET_INDEX TYPE I,
       END OF TY_CALL.

TYPES TY_T_SOURCE TYPE STANDARD TABLE OF TY_SOURCE WITH EMPTY KEY.
TYPES TY_T_NODE   TYPE STANDARD TABLE OF TY_NODE WITH EMPTY KEY.
TYPES TY_T_CALL   TYPE STANDARD TABLE OF TY_CALL WITH EMPTY KEY.
TYPES TY_T_STACK  TYPE HASHED TABLE OF STRING WITH UNIQUE KEY TABLE_LINE.
TYPES TY_T_INDEX  TYPE HASHED TABLE OF I WITH UNIQUE KEY TABLE_LINE.

DATA: GT_SOURCE TYPE TY_T_SOURCE,
      GT_NODES  TYPE TY_T_NODE,
      GT_CALLS  TYPE TY_T_CALL.

" F4用の候補。SE11の検索ヘルプは作成せず、TRDIRから都度取得する。

AT SELECTION-SCREEN ON VALUE-REQUEST FOR P_PROG.
  PERFORM F4_PROGRAM.

AT SELECTION-SCREEN.
  IF P_PROG IS NOT INITIAL AND P_PROG NA 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_'.
    MESSAGE E001(00) WITH 'プログラム名の形式が正しくありません'.
  ENDIF.

START-OF-SELECTION.
  TRANSLATE P_PROG TO UPPER CASE.
  IF P_PROG(1) <> 'Z' AND P_PROG(1) <> 'Y'.
    MESSAGE E001(00) WITH '解析対象は Z または Y で始まるレポートです'.
  ENDIF.

  SELECT SINGLE NAME
    FROM TRDIR
    WHERE NAME = @P_PROG
      AND SUBC = '1'
  INTO @DATA(LV_PROGRAM).
  IF SY-SUBRC <> 0.
    MESSAGE E001(00) WITH '指定されたReport Programは存在しません'.
  ENDIF.

  PERFORM READ_SOURCES USING P_PROG.
  IF GT_SOURCE IS INITIAL.
    MESSAGE E001(00) WITH 'ソースコードを取得できませんでした'.
  ENDIF.

  PERFORM BUILD_NODES.
  PERFORM BUILD_CALLS.
  PERFORM DISPLAY_TREE.

FORM F4_PROGRAM.
  TYPES: BEGIN OF TY_F4,
           PROGNAME TYPE PROGNAME,
           TEXT     TYPE TRDIRT-TEXT,
         END OF TY_F4.
  DATA LT_F4 TYPE STANDARD TABLE OF TY_F4 WITH EMPTY KEY.

  SELECT T~NAME AS PROGNAME, D~TEXT
    FROM TRDIR AS T
    LEFT OUTER JOIN TRDIRT AS D
      ON D~NAME = T~NAME AND D~SPRSL = @SY-LANGU
    WHERE T~SUBC = '1'
      AND ( T~NAME LIKE 'Z%' OR T~NAME LIKE 'Y%' )
    ORDER BY T~NAME
  INTO TABLE @LT_F4.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      RETFIELD    = 'PROGNAME'
      DYNPPROG    = SY-REPID
      DYNPNR      = SY-DYNNR
      DYNPROFIELD = 'P_PROG'
      VALUE_ORG   = 'S'
    TABLES
      VALUE_TAB   = LT_F4
    EXCEPTIONS
      OTHERS      = 1.
ENDFORM.

FORM READ_SOURCES USING IV_PROGRAM TYPE PROGNAME.
  DATA LT_READ_PROGRAMS TYPE TY_T_STACK.

  CLEAR GT_SOURCE.
  PERFORM READ_SOURCE_RECURSIVE USING IV_PROGRAM
                                CHANGING LT_READ_PROGRAMS.
ENDFORM.

FORM READ_SOURCE_RECURSIVE USING IV_PROGRAM TYPE PROGNAME
                           CHANGING CT_READ_PROGRAMS TYPE TY_T_STACK.
  DATA: LT_LINES    TYPE STANDARD TABLE OF STRING WITH EMPTY KEY,
        LT_INCLUDES TYPE STANDARD TABLE OF D010INC-INCLUDE WITH EMPTY KEY,
        LV_INCLUDE  TYPE PROGNAME,
        LV_PROGRAM  TYPE STRING.

  " 同一INCLUDEの重複読込、および循環INCLUDEを防止する。
  LV_PROGRAM = IV_PROGRAM.
  READ TABLE CT_READ_PROGRAMS WITH TABLE KEY TABLE_LINE = LV_PROGRAM
       TRANSPORTING NO FIELDS.
  IF SY-SUBRC = 0.
    RETURN.
  ENDIF.
  INSERT LV_PROGRAM INTO TABLE CT_READ_PROGRAMS.

  " READ REPORTはソースを参照するだけで、対象プログラムは変更しない。
  READ REPORT IV_PROGRAM INTO LT_LINES.
  LOOP AT LT_LINES INTO DATA(LV_LINE).
    APPEND VALUE #( PROGRAM = IV_PROGRAM LINE = LV_LINE ) TO GT_SOURCE.
  ENDLOOP.

  " D010INCから直接INCLUDEを取り出し、下位INCLUDEも再帰的に読み込む。
  SELECT INCLUDE FROM D010INC
    WHERE MASTER = @IV_PROGRAM
    INTO TABLE @LT_INCLUDES.
  LOOP AT LT_INCLUDES INTO LV_INCLUDE.
    PERFORM READ_SOURCE_RECURSIVE USING LV_INCLUDE
                                  CHANGING CT_READ_PROGRAMS.
  ENDLOOP.
ENDFORM.

FORM BUILD_NODES.
  DATA: LV_NAME TYPE STRING,
        LV_KIND TYPE CHAR12,
        LV_NO   TYPE I.
  CLEAR GT_NODES.

  " FORM、METHOD～ENDMETHOD、MODULE、FUNCTIONを処理単位として登録する。
  LOOP AT GT_SOURCE ASSIGNING FIELD-SYMBOL(<S>).
    LV_NAME = ``.
    LV_KIND = ``.
    FIND REGEX '^\s*FORM\s+([A-Za-z0-9_]+)' IN <S>-LINE IGNORING CASE
      SUBMATCHES LV_NAME.
    IF SY-SUBRC = 0.
      LV_KIND = 'FORM'.
    ELSE.
      FIND REGEX '^\s*METHOD\s+([A-Za-z0-9_]+)' IN <S>-LINE IGNORING CASE
        SUBMATCHES LV_NAME.
      IF SY-SUBRC = 0.
        LV_KIND = 'METHOD'.
      ELSE.
        FIND REGEX '^\s*MODULE\s+([A-Za-z0-9_]+)' IN <S>-LINE IGNORING CASE
          SUBMATCHES LV_NAME.
        IF SY-SUBRC = 0.
          LV_KIND = 'MODULE'.
        ELSE.
          FIND REGEX '^\s*FUNCTION\s+([A-Za-z0-9_]+)' IN <S>-LINE IGNORING CASE
            SUBMATCHES LV_NAME.
          IF SY-SUBRC = 0.
            LV_KIND = 'FUNCTION'.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
    IF LV_KIND IS NOT INITIAL.
      LV_NO = LINES( GT_NODES ) + 1.
      DATA(LV_DESCRIPTION) = ``.
      PERFORM GET_COMMENT USING SY-TABIX CHANGING LV_DESCRIPTION.
      APPEND VALUE #( INDEX = LV_NO KIND = LV_KIND NAME = LV_NAME
                      DESCRIPTION = LV_DESCRIPTION PROGRAM = <S>-PROGRAM
                      FIRST_LINE = SY-TABIX ) TO GT_NODES.
    ENDIF.
  ENDLOOP.

  " 各単位の終端は次の単位の直前（またはソース末尾）とする。
  LOOP AT GT_NODES ASSIGNING FIELD-SYMBOL(<N>).
    <N>-LAST_LINE = LINES( GT_SOURCE ).
    READ TABLE GT_NODES INDEX SY-TABIX + 1 INTO DATA(LS_NEXT).
    IF SY-SUBRC = 0.
      <N>-LAST_LINE = LS_NEXT-FIRST_LINE - 1.
    ENDIF.

    " 各処理単位の終端を明示的なEND文で確定する。
    DATA LV_END_PATTERN TYPE STRING.
    CASE <N>-KIND.
      WHEN 'FORM'.
        LV_END_PATTERN = '^\s*ENDFORM\s*\.'.
      WHEN 'METHOD'.
        LV_END_PATTERN = '^\s*ENDMETHOD\s*\.'.
      WHEN 'MODULE'.
        LV_END_PATTERN = '^\s*ENDMODULE\s*\.'.
      WHEN 'FUNCTION'.
        LV_END_PATTERN = '^\s*ENDFUNCTION\s*\.'.
    ENDCASE.
    IF LV_END_PATTERN IS NOT INITIAL.
      DO ( <N>-LAST_LINE - <N>-FIRST_LINE + 1 ) TIMES.
        DATA(LV_END_POS) = <N>-FIRST_LINE + SY-INDEX - 1.
        READ TABLE GT_SOURCE INDEX LV_END_POS INTO DATA(LS_END_LINE).
        IF SY-SUBRC <> 0 OR LS_END_LINE-PROGRAM <> <N>-PROGRAM.
          EXIT.
        ENDIF.
        FIND REGEX LV_END_PATTERN IN LS_END_LINE-LINE IGNORING CASE.
        IF SY-SUBRC = 0.
          <N>-LAST_LINE = LV_END_POS.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.
  ENDLOOP.

  " レポートイベントを開始地点として追加する。
  LOOP AT GT_SOURCE ASSIGNING <S> WHERE PROGRAM = P_PROG.
    FIND REGEX '^\s*(INITIALIZATION|START-OF-SELECTION|END-OF-SELECTION|TOP-OF-PAGE|AT\s+SELECTION-SCREEN)\s*\.'
      IN <S>-LINE IGNORING CASE SUBMATCHES LV_NAME.
    IF SY-SUBRC = 0.
      LV_NO = LINES( GT_NODES ) + 1.
      DATA(LV_EVENT_END) = SY-TABIX.
      DO.
        LV_EVENT_END += 1.
        READ TABLE GT_SOURCE INDEX LV_EVENT_END INTO DATA(LS_AFTER_EVENT).
        IF SY-SUBRC <> 0 OR LS_AFTER_EVENT-PROGRAM <> P_PROG.
          LV_EVENT_END -= 1.
          EXIT.
        ENDIF.
        FIND REGEX '^\s*(FORM|METHOD|MODULE|FUNCTION|INITIALIZATION|START-OF-SELECTION|END-OF-SELECTION|TOP-OF-PAGE|AT\s+SELECTION-SCREEN)\b'
          IN LS_AFTER_EVENT-LINE IGNORING CASE.
        IF SY-SUBRC = 0.
          LV_EVENT_END -= 1.
          EXIT.
        ENDIF.
      ENDDO.
      CLEAR LV_DESCRIPTION.
      PERFORM GET_COMMENT USING SY-TABIX CHANGING LV_DESCRIPTION.
      APPEND VALUE #( INDEX = LV_NO KIND = 'EVENT' NAME = LV_NAME
                      DESCRIPTION = LV_DESCRIPTION PROGRAM = <S>-PROGRAM
                      FIRST_LINE = SY-TABIX LAST_LINE = LV_EVENT_END ) TO GT_NODES.
    ENDIF.
  ENDLOOP.
ENDFORM.

FORM BUILD_CALLS.
  DATA: LV_TARGET TYPE STRING,
        LV_KIND   TYPE CHAR12.
  CLEAR GT_CALLS.
  LOOP AT GT_NODES INTO DATA(LS_NODE).
    DO ( LS_NODE-LAST_LINE - LS_NODE-FIRST_LINE ) TIMES.
      DATA(LV_POS) = LS_NODE-FIRST_LINE + SY-INDEX.
      READ TABLE GT_SOURCE INDEX LV_POS INTO DATA(LS_SOURCE).
      IF SY-SUBRC <> 0.
        CONTINUE.
      ENDIF.
      " 行全体がコメントなら解析しない（文字列内の擬似呼出しも抑制する）。
      IF LS_SOURCE-LINE IS NOT INITIAL AND LS_SOURCE-LINE+0(1) = '*'.
        CONTINUE.
      ENDIF.
      CLEAR: LV_TARGET, LV_KIND.
      FIND REGEX '\bPERFORM\s+([A-Za-z0-9_]+)' IN LS_SOURCE-LINE IGNORING CASE
        SUBMATCHES LV_TARGET.
      IF SY-SUBRC = 0.
        LV_KIND = 'FORM'.
      ELSE.
        FIND REGEX '\bCALL\s+FUNCTION\s+''([A-Za-z0-9_]+)''' IN LS_SOURCE-LINE IGNORING CASE
          SUBMATCHES LV_TARGET.
        IF SY-SUBRC = 0.
          LV_KIND = 'FUNCTION'.
        ELSE.
          DATA LV_ARROW TYPE STRING.
          FIND REGEX '(->|=>)\s*([A-Za-z0-9_]+)' IN LS_SOURCE-LINE
            IGNORING CASE SUBMATCHES LV_ARROW LV_TARGET.
          IF SY-SUBRC = 0.
            LV_KIND = 'METHOD'.
          ENDIF.
        ENDIF.
      ENDIF.
      IF LV_KIND IS INITIAL.
        CONTINUE.
      ENDIF.
      DATA(LV_TARGET_INDEX) = 0.
      LOOP AT GT_NODES INTO DATA(LS_TARGET)
           WHERE KIND = LV_KIND AND NAME = LV_TARGET.
        LV_TARGET_INDEX = LS_TARGET-INDEX.
        EXIT.
      ENDLOOP.
      APPEND VALUE #( CALLER_INDEX = LS_NODE-INDEX CALL_KIND = LV_KIND
                      TARGET_NAME = LV_TARGET TARGET_INDEX = LV_TARGET_INDEX ) TO GT_CALLS.
    ENDDO.
  ENDLOOP.
ENDFORM.

FORM DISPLAY_TREE.
  DATA: LV_ROOT_NO TYPE I,
        LV_ROOT_NUMBER TYPE STRING,
        LV_ROOT_NUMBER_CHAR TYPE C LENGTH 10,
        LT_STACK   TYPE TY_T_STACK,
        LT_DISPLAYED TYPE TY_T_INDEX.
  WRITE: / '解析対象:', P_PROG, /, '処理構造'.
  ULINE.
  LOOP AT GT_NODES INTO DATA(LS_ROOT) WHERE KIND = 'EVENT'.
    LV_ROOT_NO += 1.
    CLEAR LT_STACK.
    CLEAR LV_ROOT_NUMBER.
    WRITE LV_ROOT_NO TO LV_ROOT_NUMBER_CHAR.
    CONDENSE LV_ROOT_NUMBER_CHAR NO-GAPS.
    LV_ROOT_NUMBER = LV_ROOT_NUMBER_CHAR.
    PERFORM WRITE_NODE USING LS_ROOT-INDEX LV_ROOT_NUMBER 0
                       CHANGING LT_STACK LT_DISPLAYED.
  ENDLOOP.
  IF LV_ROOT_NO = 0.
    WRITE: / '開始イベントが見つかりませんでした。'.
  ENDIF.

  " イベントから呼び出されない処理単位も、解析結果から漏らさず表示する。
  WRITE: /, / '未呼出しの処理単位'.
  ULINE.
  LOOP AT GT_NODES INTO LS_ROOT WHERE KIND <> 'EVENT'.
    READ TABLE LT_DISPLAYED WITH TABLE KEY TABLE_LINE = LS_ROOT-INDEX
         TRANSPORTING NO FIELDS.
    IF SY-SUBRC = 0.
      CONTINUE.
    ENDIF.
    LV_ROOT_NO += 1.
    WRITE LV_ROOT_NO TO LV_ROOT_NUMBER_CHAR.
    CONDENSE LV_ROOT_NUMBER_CHAR NO-GAPS.
    LV_ROOT_NUMBER = LV_ROOT_NUMBER_CHAR.
    CLEAR LT_STACK.
    PERFORM WRITE_NODE USING LS_ROOT-INDEX LV_ROOT_NUMBER 0
                       CHANGING LT_STACK LT_DISPLAYED.
  ENDLOOP.
ENDFORM.

FORM WRITE_NODE USING IV_INDEX TYPE I IV_NUMBER TYPE STRING IV_LEVEL TYPE I
                CHANGING CT_STACK TYPE TY_T_STACK CT_DISPLAYED TYPE TY_T_INDEX.
  DATA: LV_KEY TYPE STRING,
        LV_DISPLAY TYPE STRING,
        LV_CHILD_NO TYPE I,
        LV_CHILD_NO_CHAR TYPE C LENGTH 10,
        LV_CHILD_NUMBER TYPE STRING,
        LV_NEXT_LEVEL TYPE I.
  READ TABLE GT_NODES INDEX IV_INDEX INTO DATA(LS_NODE).
  IF SY-SUBRC <> 0.
    RETURN.
  ENDIF.
  CONCATENATE LS_NODE-KIND ':' LS_NODE-NAME INTO LV_KEY.
  IF LINE_EXISTS( CT_STACK[ TABLE_LINE = LV_KEY ] ).
    WRITE: / IV_NUMBER, LS_NODE-NAME, '(循環参照)'.
    RETURN.
  ENDIF.
  INSERT LV_KEY INTO TABLE CT_STACK.
  INSERT IV_INDEX INTO TABLE CT_DISPLAYED.
  IF LS_NODE-DESCRIPTION IS INITIAL.
    CONCATENATE '[' LS_NODE-KIND ']' INTO LV_DISPLAY.
    WRITE: / IV_NUMBER, LS_NODE-NAME, LV_DISPLAY.
  ELSE.
    CONCATENATE '（' LS_NODE-DESCRIPTION '）' INTO LV_DISPLAY.
    WRITE: / IV_NUMBER, LS_NODE-NAME, LV_DISPLAY.
  ENDIF.

  CLEAR LV_CHILD_NO.
  LOOP AT GT_CALLS INTO DATA(LS_CALL) WHERE CALLER_INDEX = IV_INDEX.
    LV_CHILD_NO += 1.
    WRITE LV_CHILD_NO TO LV_CHILD_NO_CHAR.
    CONDENSE LV_CHILD_NO_CHAR NO-GAPS.
    CONCATENATE IV_NUMBER '-' LV_CHILD_NO_CHAR INTO LV_CHILD_NUMBER.
    LV_NEXT_LEVEL = IV_LEVEL + 1.
    IF LS_CALL-TARGET_INDEX > 0.
      PERFORM WRITE_NODE USING LS_CALL-TARGET_INDEX LV_CHILD_NUMBER LV_NEXT_LEVEL
                         CHANGING CT_STACK CT_DISPLAYED.
    ELSE.
      CONCATENATE '[' LS_CALL-CALL_KIND '] (外部または未定義)' INTO LV_DISPLAY.
      WRITE: / LV_CHILD_NUMBER, LS_CALL-TARGET_NAME,
               LV_DISPLAY.
    ENDIF.
  ENDLOOP.
  DELETE TABLE CT_STACK FROM LV_KEY.
ENDFORM.

FORM GET_COMMENT USING IV_LINE TYPE I CHANGING CV_TEXT TYPE STRING.
  DATA LV_TEXT TYPE STRING.
  " 直前5行から罫線ではないコメントを探す。
  DO 5 TIMES.
    DATA(LV_POS) = IV_LINE - SY-INDEX.
    IF LV_POS <= 0.
      EXIT.
    ENDIF.
    READ TABLE GT_SOURCE INDEX LV_POS INTO DATA(LS_PREV).
    READ TABLE GT_SOURCE INDEX IV_LINE INTO DATA(LS_CURRENT).
    IF LS_PREV-PROGRAM <> LS_CURRENT-PROGRAM.
      EXIT.
    ENDIF.
    LV_TEXT = LS_PREV-LINE.
    CONDENSE LV_TEXT.
    IF LV_TEXT IS NOT INITIAL AND
       ( LV_TEXT+0(1) = '*' OR LV_TEXT+0(1) = '"' ).
      REPLACE ALL OCCURRENCES OF REGEX '^[*" ]+' IN LV_TEXT WITH ``.
      CONDENSE LV_TEXT.
      IF LV_TEXT IS NOT INITIAL AND LV_TEXT NA '-=_*'.
        CV_TEXT = LV_TEXT.
        RETURN.
      ENDIF.
    ELSEIF LV_TEXT IS NOT INITIAL.
      EXIT.
    ENDIF.
  ENDDO.
ENDFORM.
