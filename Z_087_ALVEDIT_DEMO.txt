"---------------------------------------------------------------------*
" 項目カタログ作成
" P_EDIT がオンの場合だけ、変更対象の列に EDIT フラグを設定する
"---------------------------------------------------------------------*
FORM BUILD_FIELD_CATALOG.
  DEFINE ADD_FIELD.
    APPEND VALUE slis_fieldcat_alv(
      fieldname     = &1
      seltext_l     = &2
      outputlen     = &3
      edit          = &4
      checkbox      = &5
      ref_fieldname = &6
      ref_tabname   = &7
      qfieldname    = &8
      cfieldname    = &9 ) TO gt_fieldcat.
  END-OF-DEFINITION.

  " 照会モードでは、すべての列を編集不可として項目カタログを作成します。
  IF P_EDIT = ABAP_FALSE.
    CLEAR: P_SELE, P_DOC, P_DESC, P_QTY, P_UNIT, P_PRIC, P_AMT,
           P_CURR, P_DATE, P_TIME, P_DISC, P_ACTV, P_NOTE.
  ENDIF.

  " 項目別チェックがオンの列だけ、ALVの編集可能属性を設定します。
  ADD_FIELD 'LINE_NO'     '明細番号'  8  SPACE   SPACE      ''         ''     ''       ''.
  ADD_FIELD 'SELECTED'    '選択'      5  P_SELE 'X'         ''         ''     ''       ''.
  ADD_FIELD 'DOC_NO'      '伝票番号'  12 P_DOC  SPACE       ''         ''     ''       ''.
  ADD_FIELD 'DESCRIPTION' '摘要'      30 P_DESC SPACE       ''         ''     ''       ''.
  ADD_FIELD 'QUANTITY'    '数量'      15 P_QTY  SPACE       'MENGE'    'MARA' ''       'UNIT'.
  ADD_FIELD 'UNIT'        '単位'      5  P_UNIT SPACE       'MEINS'    'MARA' ''       ''.
  ADD_FIELD 'UNIT_PRICE'  '単価'      15 P_PRIC SPACE       ''         ''     ''       ''.
  ADD_FIELD 'AMOUNT'      '金額'      18 P_AMT  SPACE       'WRBTR'    'BSEG' ''       'CURRENCY'.
  ADD_FIELD 'CURRENCY'    '通貨'      5  P_CURR SPACE       'WAERS'    'BKPF' ''       ''.
  ADD_FIELD 'POST_DATE'   '転記日'    10 P_DATE SPACE       'BUDAT'    'BKPF' ''       ''.
  ADD_FIELD 'POST_TIME'   '転記時刻'  8  P_TIME SPACE       'CPUTM'    'BKPF' ''       ''.
  ADD_FIELD 'DISCOUNT'    '値引率(%)' 12 P_DISC SPACE       ''         ''     ''       ''.
  ADD_FIELD 'ACTIVE'      '有効'      5  P_ACTV 'X'         ''         ''     ''       ''.
  ADD_FIELD 'NOTE'        '備考'      35 P_NOTE SPACE       ''         ''     ''       ''.
ENDFORM.

"---------------------------------------------------------------------*
" ALV表示
" REUSE_ALV_GRID_DISPLAY は実行レポートだけでグリッドALVを表示できる。
"---------------------------------------------------------------------*
FORM DISPLAY_ALV.
  GS_LAYOUT-ZEBRA = ABAP_TRUE.
  GS_LAYOUT-COLWIDTH_OPTIMIZE = ABAP_TRUE.

  IF P_EDIT = ABAP_TRUE.
    GS_LAYOUT-WINDOW_TITLEBAR = '編集可能ALV（セルを変更して保存を押してください）'.
  ELSE.
    GS_LAYOUT-WINDOW_TITLEBAR = '照会ALV（表示専用）'.
  ENDIF.

  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      I_CALLBACK_PROGRAM      = SY-REPID
      I_CALLBACK_USER_COMMAND = 'USER_COMMAND'
      IS_LAYOUT               = GS_LAYOUT
      IT_FIELDCAT             = GT_FIELDCAT
      I_SAVE                  = 'A'
    TABLES
      T_OUTTAB                = GT_ALV_DATA
    EXCEPTIONS
      PROGRAM_ERROR           = 1
      OTHERS                  = 2.

  IF SY-SUBRC <> 0.
    MESSAGE 'ALVの表示中にエラーが発生しました。' TYPE 'S' DISPLAY LIKE 'E'.
  ENDIF.
ENDFORM.

"---------------------------------------------------------------------*
" ALVのユーザー操作
" 編集モードでは、保存ボタン（標準機能コード &DATA_SAVE）で変更結果を反映する。
"---------------------------------------------------------------------*
FORM USER_COMMAND USING PV_UCOMM    TYPE SY-UCOMM
                        PS_SELFIELD TYPE SLIS_SELFIELD.
  DATA LO_GRID TYPE REF TO CL_GUI_ALV_GRID.

  CASE PV_UCOMM.
    WHEN '&DATA_SAVE'.
      IF P_EDIT = ABAP_FALSE.
        MESSAGE '照会モードでは保存できません。' TYPE 'S' DISPLAY LIKE 'E'.
        RETURN.
      ENDIF.

      " 編集中のセル内容を内部テーブルへ確定する
      CALL FUNCTION 'GET_GLOBALS_FROM_SLVC_FULLSCR'
        IMPORTING
          E_GRID = LO_GRID.
      IF LO_GRID IS BOUND.
        LO_GRID->CHECK_CHANGED_DATA( ).
      ENDIF.

      " 学習用デモのため、DB更新は行わず内部テーブルの変更だけを保存する
      MESSAGE '変更内容を内部テーブルに保存しました（DB更新は行いません）。' TYPE 'S'.
      PS_SELFIELD-REFRESH = ABAP_TRUE.
  ENDCASE.
ENDFORM.
