AT SELECTION-SCREEN OUTPUT.
  " バリアント指定などで値が変更されても、常にシミュレーションに戻す。
  P_SIM = ABAP_TRUE.
  LOOP AT SCREEN.
    IF SCREEN-NAME = 'P_SIM'.
      SCREEN-INPUT = 0.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

AT SELECTION-SCREEN.
  " 一括操作ではP_SIMを除くP_で始まるチェックボックスだけを変更する。
  IF SY-UCOMM = 'ALLON' OR SY-UCOMM = 'ALLOFF'.
    LOOP AT SCREEN.
      IF SCREEN-NAME CP 'P_*' AND SCREEN-NAME <> 'P_SIM'.
        FIELD-SYMBOLS <LV_CHECKBOX> TYPE ANY.
        ASSIGN (SCREEN-NAME) TO <LV_CHECKBOX>.
        IF <LV_CHECKBOX> IS ASSIGNED.
          IF SY-UCOMM = 'ALLON'.
            <LV_CHECKBOX> = ABAP_TRUE.
          ELSE.
            <LV_CHECKBOX> = ABAP_FALSE.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDIF.

FORM ADD_SELECTED USING IV_SELECTED TYPE C
                        IV_BAPI     TYPE CHAR40
                        IV_DESC     TYPE CHAR60.
  IF IV_SELECTED = ABAP_TRUE.
    CLEAR GS_RESULT.
    GS_RESULT-BAPI_NAME = IV_BAPI.
    GS_RESULT-DESCRIPTION = IV_DESC.
    APPEND GS_RESULT TO GT_RESULT.
  ENDIF.
ENDFORM.

FORM EXECUTE_SELECTED_BAPIS.
  PERFORM READ_TEST_MASTER_DATA.
  LOOP AT GT_RESULT INTO GS_RESULT.
    GS_RESULT-RESULT = 'シミュレーション'.
    GS_RESULT-MESSAGE_TYPE = 'S'.
    GS_RESULT-MESSAGE = 'テストモード固定。BAPI呼出しはコメントアウト中です'.
    MODIFY GT_RESULT FROM GS_RESULT INDEX SY-TABIX.
  ENDLOOP.
ENDFORM.

FORM READ_TEST_MASTER_DATA.
  " 現行環境の登録済み標準データを参照し、個別BAPI実装時の引数候補とする。
  SELECT SINGLE VBELN KUNNR VKORG VTWEG SPART
    INTO (GV_VBELN, GV_KUNNR, GV_VKORG, GV_VTWEG, GV_SPART)
    FROM VBAK.
  SELECT SINGLE MATNR WERKS
    INTO (GV_MATNR, GV_WERKS)
    FROM MARC.
  SELECT SINGLE BUKRS INTO GV_BUKRS FROM T001.
  SELECT SINGLE EBELN INTO GV_EBELN FROM EKKO.
  SELECT SINGLE BANFN INTO GV_BANFN FROM EBAN.
ENDFORM.

" BAPI個別FORMの実装例
" 標準テーブルから取得したGV_*を、該当BAPIの構造へ設定して使用する。
" インターフェースと必須項目を対象システムで検証するまではCALL FUNCTIONを
" コメントアウトする。BAPI_TRANSACTION_COMMITは絶対に呼び出さない。

FORM EXECUTE_SALESORDER_CREATE.
  " 受注伝票登録：GV_VKORG、GV_VTWEG、GV_SPART、GV_KUNNR、GV_MATNR、GV_WERKSを使用
  "* CALL FUNCTION 'BAPI_SALESORDER_CREATEFROMDAT2'
  "*   EXPORTING TESTRUN = ABAP_TRUE
  "*   TABLES RETURN = LT_RETURN.
ENDFORM.

FORM EXECUTE_SALESORDER_CHANGE.
  " 受注伝票変更：GV_VBELNを変更対象候補として使用
  "* CALL FUNCTION 'BAPI_SALESORDER_CHANGE'
  "*   EXPORTING SALESDOCUMENT = GV_VBELN SIMULATION = ABAP_TRUE
  "*   TABLES RETURN = LT_RETURN.
ENDFORM.

FORM EXECUTE_PO_CREATE.
  " 発注伝票登録：GV_BUKRS、GV_WERKS、GV_MATNRを購買伝票ヘッダ・明細へ設定
  "* CALL FUNCTION 'BAPI_PO_CREATE1'
  "*   TABLES RETURN = LT_RETURN.
ENDFORM.

FORM EXECUTE_GOODSMVT_CREATE.
  " 在庫移動：GV_MATNRおよびGV_WERKSを明細へ設定
  "* CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
  "*   EXPORTING GOODSMVT_HEADER = LS_HEADER GOODSMVT_CODE = LS_CODE
  "*   TABLES GOODSMVT_ITEM = LT_ITEM RETURN = LT_RETURN.
ENDFORM.

FORM EXECUTE_ACC_DOCUMENT_POST.
  " 会計伝票：GV_BUKRSをヘッダへ設定。勘定・金額はテスト専用値が必要。
  "* CALL FUNCTION 'BAPI_ACC_DOCUMENT_POST'
  "*   EXPORTING DOCUMENTHEADER = LS_HEADER
  "*   TABLES ACCOUNTGL = LT_ACCOUNTGL CURRENCYAMOUNT = LT_AMOUNT RETURN = LT_RETURN.
ENDFORM.

FORM DISPLAY_RESULT.
  ULINE.
  WRITE: / 'BAPI', 45 '処理内容', 75 '結果', 90 'MESSAGE'.
  ULINE.
  LOOP AT GT_RESULT INTO GS_RESULT.
    WRITE: / GS_RESULT-BAPI_NAME,
             45 GS_RESULT-DESCRIPTION,
             75 GS_RESULT-RESULT,
             90 GS_RESULT-MESSAGE.
  ENDLOOP.
ENDFORM.
